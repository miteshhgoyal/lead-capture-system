// ===== FILE: ./app/[slug]/page.tsx =====
import { cache } from "react";
import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { dbConnect } from "@/lib/db";
import { Partner, type IPartner } from "@/lib/models/Partner";
import { getSettings } from "@/lib/settings";
import { isReservedSlug } from "@/lib/validation/partner";
import { ensurePartnerPage } from "@/lib/page-builder/persist";
import { resolvePage, type PartnerView } from "@/components/landing/page-context";
import { LandingPage } from "@/components/landing/LandingPage";
import { resolveMeta } from "@/lib/page-builder/resolve";

interface PartnerPageProps {
  params: Promise<{ slug: string }>;
}

/** Partner + its saved page, loaded once per request (shared by metadata and the page body). */
const getPartnerPage = cache(async (slugParam: string) => {
  const slug = (slugParam || "").toLowerCase().trim();
  if (isReservedSlug(slug)) return null;

  try {
    await dbConnect();
    const partner = await Partner.findOne({ slug, isActive: true }).lean<IPartner | null>();
    if (!partner) return null;
    // Reads the saved config; generates and saves one only if this partner has none yet.
    const page = await ensurePartnerPage(partner);
    return { partner, page };
  } catch (error) {
    console.error("[getPartnerPage] Error loading partner page:", error);
    return null;
  }
});

function toPartnerView(partner: IPartner): PartnerView {
  return {
    name: partner.name,
    slug: partner.slug,
    telegramLink: partner.telegramLink,
    whatsappLink: partner.whatsappGroupLink,
    tagline: partner.tagline,
    mentorBio: partner.mentorBio,
    logoUrl: partner.logoUrl,
    theme: { primaryColor: partner.theme?.primaryColor },
    testimonials: (partner.testimonials ?? []).map((t) => ({ name: t.name, role: t.role, quote: t.quote })),
  };
}

export async function generateMetadata(props: PartnerPageProps): Promise<Metadata> {
  const params = await props.params;
  const data = await getPartnerPage(params.slug);
  const settings = await getSettings();

  if (!data) {
    return {
      title: "Partner Page Not Found",
    };
  }

  const { partner, page } = data;
  const { vars } = resolvePage({ partner: toPartnerView(partner), settings, isCustomPartner: true, plan: page });
  const { title, description } = resolveMeta(page.contentConfig.meta, vars, partner.tagline);
  const images = partner.logoUrl ? [{ url: partner.logoUrl }] : settings.logoUrl ? [{ url: settings.logoUrl }] : [];

  return {
    title,
    description,
    openGraph: {
      title,
      description,
      type: "website",
      images,
    },
    twitter: {
      card: "summary_large_image",
      title,
      description,
      images: images.map((i) => i.url),
    },
  };
}

export default async function PartnerPage(props: PartnerPageProps) {
  const params = await props.params;
  const data = await getPartnerPage(params.slug);
  const settings = await getSettings();

  if (!data) {
    notFound();
  }

  return <LandingPage partner={toPartnerView(data.partner)} settings={settings} isCustomPartner page={data.page} />;
}


// ===== FILE: ./app/admin/layout.tsx =====
import { Suspense, type ReactNode } from "react";
import { headers } from "next/headers";
import { redirect } from "next/navigation";
import type { Metadata } from "next";
import { requireAdmin } from "@/lib/admin-auth";
import { AdminShell } from "@/components/admin/AdminShell";

export const metadata: Metadata = {
  robots: { index: false, follow: false },
  title: { default: "Admin Console", template: "%s | Free Course Link Admin" },
};

async function AdminAuthGuard({ children }: { children: ReactNode }) {
  const h = await headers();
  const pathname = h.get("x-pathname") || "";

  if (pathname === "/admin/login") {
    return <>{children}</>;
  }

  try {
    await requireAdmin();
  } catch {
    redirect("/admin/login");
  }

  return <AdminShell>{children}</AdminShell>;
}

export default function AdminLayout({ children }: { children: React.ReactNode }) {
  return (
    <Suspense fallback={<div className="min-h-screen bg-bg" />}>
      <AdminAuthGuard>{children}</AdminAuthGuard>
    </Suspense>
  );
}


// ===== FILE: ./app/admin/leads/page.tsx =====
"use client";

import { useEffect, useState, useCallback } from "react";
import {
  Button,
  Card,
  Input,
} from "@/components/admin/AdminPrimitives";
import { Drawer } from "@/components/admin/Drawer";
import { ConfirmDialog } from "@/components/admin/ConfirmDialog";
import { useToast } from "@/components/admin/Toast";
import {
  CAPITAL_OPTIONS,
  type LeadStatus,
} from "@/lib/validation/lead";
import { ArrowRight, MessageCircle, Search, Trash2 } from "lucide-react";

interface LeadItem {
  _id: string;
  name: string;
  phone: string;
  countryCode: string;
  email: string;
  age: number;
  profession: string;
  capital: string;
  country: string;
  partnerSlug: string;
  partnerName: string;
  source: "partner-page" | "direct";
  utm_source?: string;
  utm_medium?: string;
  utm_campaign?: string;
  referrer?: string;
  landingUrl?: string;
  ip?: string;
  userAgent?: string;
  device?: string;
  status: LeadStatus;
  notes?: string;
  isDuplicate: boolean;
  duplicateCount: number;
  createdAt: string;
  updatedAt: string;
}

function statusSelectClass(status: LeadStatus) {
  return status === "new"
    ? "border-blue-500/30 bg-blue-500/10 text-blue-500"
    : status === "contacted"
    ? "border-amber-500/30 bg-amber-500/10 text-amber-500"
    : status === "joined"
    ? "border-up/30 bg-up/10 text-up"
    : "border-down/30 bg-down/10 text-down";
}

export default function LeadsPage() {
  const { showToast } = useToast();

  const [leads, setLeads] = useState<LeadItem[]>([]);
  const [totalLeads, setTotalLeads] = useState(0);
  const [page, setPage] = useState(1);
  const [limit] = useState(20);
  const [totalPages, setTotalPages] = useState(1);
  const [loading, setLoading] = useState(true);

  // Filters
  const [search, setSearch] = useState("");
  const [partnerFilter, setPartnerFilter] = useState("all");
  const [statusFilter, setStatusFilter] = useState("all");
  const [capitalFilter, setCapitalFilter] = useState("all");
  const [startDate, setStartDate] = useState("");
  const [endDate, setEndDate] = useState("");

  const [partnersList, setPartnersList] = useState<{ slug: string; name: string }[]>([]);

  // Selection & Bulk
  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
  const [bulkStatus, setBulkStatus] = useState<LeadStatus>("contacted");

  // Lead Detail Drawer
  const [activeLead, setActiveLead] = useState<LeadItem | null>(null);
  const [editNotes, setEditNotes] = useState("");
  const [savingNotes, setSavingNotes] = useState(false);

  // Delete Target
  const [deleteTarget, setDeleteTarget] = useState<LeadItem | null>(null);

  // Pre-select the partner when arriving from the Performance page (?partnerSlug=...)
  useEffect(() => {
    const slug = new URLSearchParams(window.location.search).get("partnerSlug");
    if (slug) setPartnerFilter(slug);
  }, []);

  // Load partners list for filter dropdown
  useEffect(() => {
    fetch("/api/admin/partners")
      .then((res) => res.json())
      .then((data) => {
        if (data.ok && Array.isArray(data.partners)) {
          setPartnersList(data.partners.map((p: any) => ({ slug: p.slug, name: p.name })));
        }
      })
      .catch(() => {});
  }, []);

  const fetchLeads = useCallback(async () => {
    setLoading(true);
    try {
      const q = new URLSearchParams({
        page: String(page),
        limit: String(limit),
      });

      if (search.trim()) q.set("q", search.trim());
      if (partnerFilter !== "all") q.set("partnerSlug", partnerFilter);
      if (statusFilter !== "all") q.set("status", statusFilter);
      if (capitalFilter !== "all") q.set("capital", capitalFilter);
      if (startDate) q.set("startDate", startDate);
      if (endDate) q.set("endDate", endDate);

      const res = await fetch(`/api/admin/leads?${q.toString()}`);
      const json = await res.json();
      if (json.ok) {
        setLeads(json.leads);
        setTotalLeads(json.pagination.total);
        setTotalPages(json.pagination.totalPages);
      }
    } catch {
      showToast("Failed to fetch leads", "error");
    } finally {
      setLoading(false);
    }
  }, [page, limit, search, partnerFilter, statusFilter, capitalFilter, startDate, endDate, showToast]);

  useEffect(() => {
    fetchLeads();
  }, [fetchLeads]);

  // Inline status change
  const handleStatusChange = async (leadId: string, newStatus: LeadStatus) => {
    try {
      const res = await fetch(`/api/admin/leads/${leadId}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ status: newStatus }),
      });
      if (res.ok) {
        setLeads((prev) =>
          prev.map((l) => (l._id === leadId ? { ...l, status: newStatus } : l))
        );
        if (activeLead && activeLead._id === leadId) {
          setActiveLead((prev) => (prev ? { ...prev, status: newStatus } : null));
        }
        showToast(`Lead status updated to "${newStatus}"`, "success");
      }
    } catch {
      showToast("Failed to update status", "error");
    }
  };

  // Save notes from detail drawer
  const handleSaveNotes = async () => {
    if (!activeLead) return;
    setSavingNotes(true);
    try {
      const res = await fetch(`/api/admin/leads/${activeLead._id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ notes: editNotes }),
      });
      if (res.ok) {
        setLeads((prev) =>
          prev.map((l) => (l._id === activeLead._id ? { ...l, notes: editNotes } : l))
        );
        setActiveLead((prev) => (prev ? { ...prev, notes: editNotes } : null));
        showToast("Notes updated", "success");
      }
    } catch {
      showToast("Failed to update notes", "error");
    } finally {
      setSavingNotes(false);
    }
  };

  // Delete single lead
  const handleDeleteLead = async () => {
    if (!deleteTarget) return;
    try {
      const res = await fetch(`/api/admin/leads/${deleteTarget._id}`, {
        method: "DELETE",
      });
      if (res.ok) {
        setLeads((prev) => prev.filter((l) => l._id !== deleteTarget._id));
        setTotalLeads((prev) => prev - 1);
        showToast("Lead deleted", "success");
      }
    } catch {
      showToast("Failed to delete lead", "error");
    } finally {
      setDeleteTarget(null);
    }
  };

  // Bulk status update
  const handleBulkStatusUpdate = async () => {
    if (selectedIds.size === 0) return;
    try {
      const res = await fetch("/api/admin/leads", {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          ids: Array.from(selectedIds),
          status: bulkStatus,
        }),
      });
      const data = await res.json();
      if (res.ok && data.ok) {
        showToast(`Updated ${data.modifiedCount} leads to "${bulkStatus}"`, "success");
        setSelectedIds(new Set());
        fetchLeads();
      }
    } catch {
      showToast("Bulk update failed", "error");
    }
  };

  // Bulk delete
  const handleBulkDelete = async () => {
    if (selectedIds.size === 0) return;
    if (!confirm(`Are you sure you want to delete ${selectedIds.size} selected leads?`)) return;

    try {
      const res = await fetch("/api/admin/leads", {
        method: "DELETE",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ ids: Array.from(selectedIds) }),
      });
      const data = await res.json();
      if (res.ok && data.ok) {
        showToast(`Deleted ${data.deletedCount} leads`, "success");
        setSelectedIds(new Set());
        fetchLeads();
      }
    } catch {
      showToast("Bulk delete failed", "error");
    }
  };

  const toggleSelectAll = () => {
    if (selectedIds.size === leads.length && leads.length > 0) {
      setSelectedIds(new Set());
    } else {
      setSelectedIds(new Set(leads.map((l) => l._id)));
    }
  };

  const toggleSelectOne = (id: string) => {
    setSelectedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  // Export CSV url
  const getExportUrl = () => {
    const q = new URLSearchParams();
    if (search.trim()) q.set("q", search.trim());
    if (partnerFilter !== "all") q.set("partnerSlug", partnerFilter);
    if (statusFilter !== "all") q.set("status", statusFilter);
    if (capitalFilter !== "all") q.set("capital", capitalFilter);
    if (startDate) q.set("startDate", startDate);
    if (endDate) q.set("endDate", endDate);
    return `/api/admin/leads/export?${q.toString()}`;
  };

  const openDetailDrawer = (lead: LeadItem) => {
    setActiveLead(lead);
    setEditNotes(lead.notes || "");
  };

  const allSelected = leads.length > 0 && selectedIds.size === leads.length;

  const pagination = (
    <div className="flex flex-col items-center justify-between gap-4 p-4 sm:flex-row">
      <div className="text-xs text-mute">
        Showing {totalLeads === 0 ? 0 : (page - 1) * limit + 1} to {Math.min(page * limit, totalLeads)} of{" "}
        {totalLeads.toLocaleString()} leads
      </div>

      <div className="flex items-center gap-2">
        <button
          disabled={page <= 1}
          onClick={() => setPage((p) => Math.max(1, p - 1))}
          className="rounded-lg border border-hairline bg-card px-3 py-1.5 text-xs font-semibold text-ink transition hover:bg-bg-alt disabled:opacity-40"
        >
          Previous
        </button>
        <span className="text-xs text-mute">
          Page {page} of {Math.max(1, totalPages)}
        </span>
        <button
          disabled={page >= totalPages}
          onClick={() => setPage((p) => p + 1)}
          className="rounded-lg border border-hairline bg-card px-3 py-1.5 text-xs font-semibold text-ink transition hover:bg-bg-alt disabled:opacity-40"
        >
          Next
        </button>
      </div>
    </div>
  );

  return (
    <div className="space-y-6">
      {/* Top Header & Export */}
      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="text-2xl font-bold tracking-tight text-ink">Leads Data Table</h1>
          <p className="mt-1 text-sm text-mute">
            {totalLeads.toLocaleString()} total captured webinar registrations across all sources.
          </p>
        </div>

        <div className="flex items-center gap-3">
          <a
            href={getExportUrl()}
            download
            className="inline-flex items-center gap-2 rounded-xl border border-hairline bg-card px-4 py-2.5 text-xs font-semibold text-ink shadow-sm transition hover:bg-bg-alt"
          >
            <svg className="size-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
              <path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4" />
              <polyline points="7 10 12 15 17 10" />
              <line x1="12" y1="15" x2="12" y2="3" />
            </svg>
            Export to CSV
          </a>
        </div>
      </div>

      {/* Filters Bar */}
      <Card className="p-4 space-y-4">
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 md:grid-cols-5">
          {/* Search */}
          <div className="md:col-span-2">
            <label className="mb-1 block text-[11px] font-semibold text-mute">Search</label>
            <Input
              placeholder="Search by name, phone, email..."
              value={search}
              onChange={(e) => {
                setSearch(e.target.value);
                setPage(1);
              }}
            />
          </div>

          {/* Partner Filter */}
          <div>
            <label className="mb-1 block text-[11px] font-semibold text-mute">Partner</label>
            <select
              value={partnerFilter}
              onChange={(e) => {
                setPartnerFilter(e.target.value);
                setPage(1);
              }}
              className="w-full rounded-xl border border-hairline bg-bg px-3 py-2 text-xs text-ink"
            >
              <option value="all">All Partners</option>
              <option value="direct">Direct (No Partner)</option>
              {partnersList.map((p) => (
                <option key={p.slug} value={p.slug}>
                  {p.name} (/{p.slug})
                </option>
              ))}
            </select>
          </div>

          {/* Status Filter */}
          <div>
            <label className="mb-1 block text-[11px] font-semibold text-mute">Status</label>
            <select
              value={statusFilter}
              onChange={(e) => {
                setStatusFilter(e.target.value);
                setPage(1);
              }}
              className="w-full rounded-xl border border-hairline bg-bg px-3 py-2 text-xs text-ink"
            >
              <option value="all">All Statuses</option>
              <option value="new">New</option>
              <option value="contacted">Contacted</option>
              <option value="joined">Joined</option>
              <option value="not-interested">Not Interested</option>
            </select>
          </div>

          {/* Capital Filter */}
          <div>
            <label className="mb-1 block text-[11px] font-semibold text-mute">Capital</label>
            <select
              value={capitalFilter}
              onChange={(e) => {
                setCapitalFilter(e.target.value);
                setPage(1);
              }}
              className="w-full rounded-xl border border-hairline bg-bg px-3 py-2 text-xs text-ink"
            >
              <option value="all">All Capital Ranges</option>
              {CAPITAL_OPTIONS.map((c) => (
                <option key={c} value={c}>
                  {c}
                </option>
              ))}
            </select>
          </div>
        </div>

        {/* Date Filters & Clear */}
        <div className="flex flex-wrap items-center justify-between gap-3 border-t border-hairline pt-3">
          <div className="flex flex-wrap items-center gap-2">
            <span className="text-xs text-mute">Date:</span>
            <input
              type="date"
              value={startDate}
              onChange={(e) => {
                setStartDate(e.target.value);
                setPage(1);
              }}
              className="min-w-0 rounded-lg border border-hairline bg-bg px-2 py-1 text-xs text-ink"
            />
            <span className="text-xs text-mute">to</span>
            <input
              type="date"
              value={endDate}
              onChange={(e) => {
                setEndDate(e.target.value);
                setPage(1);
              }}
              className="min-w-0 rounded-lg border border-hairline bg-bg px-2 py-1 text-xs text-ink"
            />
          </div>

          {(search || partnerFilter !== "all" || statusFilter !== "all" || capitalFilter !== "all" || startDate || endDate) && (
            <button
              onClick={() => {
                setSearch("");
                setPartnerFilter("all");
                setStatusFilter("all");
                setCapitalFilter("all");
                setStartDate("");
                setEndDate("");
                setPage(1);
              }}
              className="text-xs font-semibold text-down hover:underline"
            >
              Reset Filters
            </button>
          )}
        </div>
      </Card>

      {/* Bulk Actions Bar */}
      {selectedIds.size > 0 && (
        <div className="flex flex-col gap-3 rounded-2xl border border-accent/30 bg-accent/10 px-4 py-3 sm:flex-row sm:items-center sm:justify-between sm:px-5">
          <div className="flex items-center gap-2 text-xs font-semibold text-ink">
            <span className="flex size-5 items-center justify-center rounded-full bg-accent text-accent-ink text-[10px]">
              {selectedIds.size}
            </span>
            <span>Selected</span>
          </div>

          <div className="flex flex-wrap items-center gap-2 sm:gap-3">
            <select
              value={bulkStatus}
              onChange={(e) => setBulkStatus(e.target.value as LeadStatus)}
              className="rounded-lg border border-hairline bg-bg px-2.5 py-1.5 text-xs text-ink"
            >
              <option value="new">Mark New</option>
              <option value="contacted">Mark Contacted</option>
              <option value="joined">Mark Joined</option>
              <option value="not-interested">Mark Not Interested</option>
            </select>
            <button
              onClick={handleBulkStatusUpdate}
              className="rounded-lg bg-accent px-3 py-1.5 text-xs font-semibold text-accent-ink shadow-sm"
            >
              Apply Status
            </button>
            <button
              onClick={handleBulkDelete}
              className="rounded-lg bg-down px-3 py-1.5 text-xs font-semibold text-white shadow-sm"
            >
              Delete Selected
            </button>
          </div>
        </div>
      )}

      {/* ───────── Mobile: card list ───────── */}
      <div className="space-y-3 md:hidden">
        {loading ? (
          <div className="rounded-2xl border border-hairline bg-card px-6 py-12 text-center text-sm text-mute">
            Loading leads...
          </div>
        ) : leads.length === 0 ? (
          <div className="rounded-2xl border border-hairline bg-card px-6 py-12 text-center text-sm text-mute">
            No leads found matching your criteria.
          </div>
        ) : (
          <>
            <label className="flex items-center gap-2 px-1 text-xs font-semibold text-mute">
              <input
                type="checkbox"
                checked={allSelected}
                onChange={toggleSelectAll}
                className="size-4 rounded border-hairline"
              />
              Select all on this page
            </label>

            {leads.map((lead) => {
              const whatsappClean = lead.phone.replace(/[^\d]/g, "");
              const isSelected = selectedIds.has(lead._id);

              return (
                <div
                  key={lead._id}
                  className={`rounded-2xl border bg-card p-4 ${
                    isSelected ? "border-accent/40 bg-accent/5" : "border-hairline"
                  }`}
                >
                  <div className="flex items-start gap-3">
                    <input
                      type="checkbox"
                      checked={isSelected}
                      onChange={() => toggleSelectOne(lead._id)}
                      className="mt-1 size-4 shrink-0 rounded border-hairline"
                      aria-label={`Select ${lead.name}`}
                    />
                    <div className="min-w-0 flex-1">
                      <div className="flex flex-wrap items-center gap-1.5">
                        <span className="break-words font-semibold text-ink">{lead.name}</span>
                        {lead.isDuplicate && (
                          <span className="rounded bg-amber-500/15 px-1 py-0.5 text-[9px] font-bold text-amber-500">
                            DUP
                          </span>
                        )}
                      </div>
                      <div className="mt-0.5 text-[11px] text-mute">
                        {new Date(lead.createdAt).toLocaleDateString()}{" "}
                        {new Date(lead.createdAt).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
                      </div>
                    </div>
                    <div className="flex shrink-0 items-center">
                      <button
                        onClick={() => openDetailDrawer(lead)}
                        className="rounded-lg p-2 text-mute hover:bg-bg-alt hover:text-ink"
                        aria-label="View lead details"
                      >
                        <Search className="size-4" aria-hidden />
                      </button>
                      <button
                        onClick={() => setDeleteTarget(lead)}
                        className="rounded-lg p-2 text-mute hover:bg-down/10 hover:text-down"
                        aria-label="Delete lead"
                      >
                        <Trash2 className="size-4" aria-hidden />
                      </button>
                    </div>
                  </div>

                  <div className="mt-3 space-y-2 text-xs">
                    <div className="flex items-center gap-2 font-mono text-ink">
                      <span className="break-all">{lead.phone}</span>
                      <a
                        href={`https://wa.me/${whatsappClean}`}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="shrink-0 text-up"
                        aria-label="Chat on WhatsApp"
                      >
                        <MessageCircle className="size-4" aria-hidden />
                      </a>
                    </div>
                    <div className="break-all text-mute">{lead.email}</div>
                    <div className="flex flex-wrap items-center gap-x-3 gap-y-1 text-mute">
                      <span>
                        {lead.age} yrs · {lead.profession}

"use client";

import { useState, type FormEvent } from "react";
import { useRouter } from "next/navigation";
import { Button, Field, Input } from "@/components/admin/AdminPrimitives";
import { ThemeToggle } from "@/components/landing/ThemeToggle";
import { ArrowRight } from "lucide-react";

export default function AdminLoginPage() {
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault();
    setBusy(true);
    setError("");

    try {
      const res = await fetch("/api/admin/login", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email: email.trim(), password }),
      });

      const data = (await res.json()) as { ok?: boolean; message?: string; error?: string };

      if (res.ok && data.ok) {
        router.push("/admin");
        router.refresh();
      } else {
        if (res.status === 429) {
          setError(data.message ?? "Too many attempts. Please try again later.");
        } else {
          setError(data.message ?? "Invalid email or password.");
        }
      }
    } catch {
      setError("Network error. Please try again.");
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="relative min-h-screen flex items-center justify-center bg-bg p-4 font-sans overflow-hidden">
      {/* Background Animated Blobs & Grid */}
      <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden="true">
        <div className="f-blob a absolute -top-40 -left-40 size-96 rounded-full opacity-30" />
        <div className="f-blob b absolute -bottom-40 -right-40 size-96 rounded-full opacity-25" />
        <div className="f-grid absolute inset-0 opacity-40 [mask-image:radial-gradient(circle_at_center,black_40%,transparent_80%)]" />
      </div>

      <div className="absolute top-4 right-4 z-20">
        <ThemeToggle />
      </div>

      <div className="relative w-full max-w-sm rounded-3xl border border-hairline bg-card/95 backdrop-blur-xl p-8 shadow-2xl">
        <div className="text-center mb-6">
          <div className="inline-flex size-14 items-center justify-center rounded-2xl bg-gradient-to-br from-accent to-accent-2 text-accent-ink font-bold text-2xl mb-3 shadow-[0_8px_24px_-8px_color-mix(in_oklab,var(--accent)_60%,transparent)]">
            OS
          </div>
          <h1 className="text-xl font-bold tracking-tight text-ink">Lead Capture Admin</h1>
          <p className="mt-1 text-meta text-mute">
            Sign in with administrator credentials
          </p>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <Field label="Email Address" required>
            <Input
              id="email"
              type="email"
              required
              autoComplete="email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              placeholder="admin@freecourselink.com"
            />
          </Field>

          <Field label="Password" required>
            <Input
              id="password"
              type="password"
              required
              autoComplete="current-password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="••••••••••••"
            />
          </Field>

          {error && (
            <div
              role="alert"
              className="rounded-xl border border-down/30 bg-down/10 p-3 text-meta font-medium text-down"
            >
              {error}
            </div>
          )}

          <Button
            type="submit"
            disabled={busy}
            className="w-full justify-center bg-accent text-accent-ink font-semibold py-3"
          >
            {busy ? "Signing in..." : <>Sign In <ArrowRight className="ml-1 inline size-4" aria-hidden /></>}

        <p className="mt-6 text-center text-xs text-mute">

              );
            })}
          </div>
        )}
      </Card>

      {/* Conversion by Status & Top Partners */}
      <div className="grid grid-cols-1 gap-6 lg:grid-cols-12">
        {/* Status Distribution */}
        <div className="lg:col-span-5">
          <Card className="p-6 h-full">
            <h3 className="text-base font-bold text-ink">Leads by Status</h3>
            <p className="text-xs text-mute">Pipeline conversion distribution</p>

            <div className="mt-6 space-y-3">
              {[
                { key: "new", label: "New Leads", color: "bg-blue-500", text: "text-blue-500" },
                { key: "contacted", label: "Contacted", color: "bg-amber-500", text: "text-amber-500" },
                { key: "joined", label: "Joined Webinar", color: "bg-up", text: "text-up" },
                { key: "not-interested", label: "Not Interested", color: "bg-down", text: "text-down" },
              ].map((s) => {
                const count =
                  data?.statusBreakdown.find((x) => x.status === s.key)?.count || 0;
                const total = data?.metrics.totalLeads || 1;
                const pct = Math.round((count / total) * 100);

                return (
                  <div key={s.key} className="space-y-1.5">
                    <div className="flex items-center justify-between text-xs">
                      <span className="font-medium text-ink">{s.label}</span>
                      <span className="font-semibold text-mute">
                        {count.toLocaleString()} ({pct}%)
                      </span>
                    </div>
                    <div className="h-2 w-full overflow-hidden rounded-full bg-hairline">
                      <div
                        style={{ width: `${pct}%` }}
                        className={`h-full rounded-full ${s.color}`}
                      />
                    </div>
                  </div>
                );
              })}
            </div>
          </Card>
        </div>

        {/* Top Partners */}
        <div className="lg:col-span-7">
          <Card className="p-6 h-full">
            <div className="flex items-center justify-between pb-3">
              <div>
                <h3 className="text-base font-bold text-ink">Top Partners by Leads</h3>
                <p className="text-xs text-mute">Highest performing referral partners</p>
              </div>
              <Link
                href="/admin/partners"
                className="text-xs font-semibold text-accent hover:underline"
              >
                Manage Partners <ArrowRight className="ml-1 inline size-3.5" aria-hidden />
              </Link>
            </div>

            {!data?.partnerBreakdown || data.partnerBreakdown.length === 0 ? (
              <div className="py-12 text-center text-sm text-mute">
                No partner data available yet.
              </div>
            ) : (
              <div className="mt-4 divide-y divide-hairline">
                {data.partnerBreakdown.map((p, i) => (
                  <div key={p.slug} className="flex items-center justify-between py-3">
                    <div className="flex items-center gap-3">
                      <span className="flex size-6 items-center justify-center rounded-full bg-hairline text-[11px] font-bold text-mute">
                        {i + 1}
                      </span>
                      <div>
                        <div className="text-sm font-semibold text-ink">{p.name}</div>
                        <div className="text-xs text-mute">/{p.slug}</div>
                      </div>
                    </div>
                    <div className="text-right">
                      <div className="text-sm font-bold text-ink">
                        {p.count.toLocaleString()} leads
                      </div>
                      <div className="text-[11px] text-mute">{p.sharePct}% share</div>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </Card>
        </div>
      </div>

      {/* Breakdowns: Capital, Profession, Country */}
      <div className="grid grid-cols-1 gap-6 md:grid-cols-3">
        {/* By Capital */}
        <Card className="p-5">
          <h4 className="text-sm font-bold text-ink">Trading Capital</h4>
          <p className="text-[11px] text-mute">Declared investment range</p>
          <div className="mt-4 space-y-2.5">
            {(data?.capitalBreakdown || []).map((c) => (
              <div key={c.capital} className="flex items-center justify-between text-xs">
                <span className="text-mute truncate max-w-[180px]">{c.capital}</span>
                <span className="font-semibold text-ink">{c.count}</span>
              </div>
            ))}
          </div>
        </Card>

        {/* By Profession */}
        <Card className="p-5">
          <h4 className="text-sm font-bold text-ink">Profession</h4>
          <p className="text-[11px] text-mute">Background breakdown</p>
          <div className="mt-4 space-y-2.5">
            {(data?.professionBreakdown || []).map((p) => (
              <div key={p.profession} className="flex items-center justify-between text-xs">
                <span className="text-mute">{p.profession}</span>
                <span className="font-semibold text-ink">{p.count}</span>
              </div>
            ))}
          </div>
        </Card>

        {/* By Country */}
        <Card className="p-5">
          <h4 className="text-sm font-bold text-ink">Top Countries</h4>
          <p className="text-[11px] text-mute">Geographic distribution</p>
          <div className="mt-4 space-y-2.5">
            {(data?.countryBreakdown || []).map((c) => (
              <div key={c.country} className="flex items-center justify-between text-xs">
                <span className="text-mute">{c.country}</span>
                <span className="font-semibold text-ink">{c.count}</span>
              </div>
            ))}
          </div>
        </Card>
      </div>
    </div>
  );
}

"use client";

import { useEffect, useState, useTransition, useCallback, type ReactNode } from "react";
import {
  Button,
  Card,
  Input,
  Field,
} from "@/components/admin/AdminPrimitives";
import { Drawer } from "@/components/admin/Drawer";
import { ConfirmDialog } from "@/components/admin/ConfirmDialog";
import { useToast } from "@/components/admin/Toast";
import { slugify, isReservedSlug } from "@/lib/validation/partner";
import { getReadableTextColor } from "@/lib/theme-utils";
import { ArrowRight, Check, Copy, ExternalLink, Plus, Shuffle, Trash2, X } from "lucide-react";

interface TestimonialInput {
  name: string;
  role?: string;
  quote: string;
}

const MAX_TESTIMONIALS = 6;

interface PartnerItem {
  _id: string;
  name: string;
  slug: string;
  telegramLink: string;
  whatsappGroupLink?: string;
  zoomLink?: string;
  isActive: boolean;
  tagline?: string;
  mentorBio?: string;
  logoUrl?: string;
  testimonials?: TestimonialInput[];
  theme: {
    primaryColor: string;
  };
  leadsCount: number;
  createdAt: string;
}

export default function PartnersPage() {
  const { showToast } = useToast();
  const [partners, setPartners] = useState<PartnerItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [drawerOpen, setDrawerOpen] = useState(false);
  const [editingPartner, setEditingPartner] = useState<PartnerItem | null>(null);
  const [deleteTarget, setDeleteTarget] = useState<PartnerItem | null>(null);
  const [regenTarget, setRegenTarget] = useState<PartnerItem | null>(null);
  const [savedSuccessUrl, setSavedSuccessUrl] = useState<string | null>(null);

  // Form state
  const [name, setName] = useState("");
  const [slug, setSlug] = useState("");
  const [slugStatus, setSlugStatus] = useState<"idle" | "checking" | "available" | "unavailable">("idle");
  const [slugReason, setSlugReason] = useState("");
  const [telegramLink, setTelegramLink] = useState("");
  const [whatsappLink, setWhatsappLink] = useState("");
  const [zoomLink, setZoomLink] = useState("");
  const [tagline, setTagline] = useState("");
  const [mentorBio, setMentorBio] = useState("");
  const [testimonials, setTestimonials] = useState<TestimonialInput[]>([]);
  const [logoUrl, setLogoUrl] = useState("");
  const [primaryColor, setPrimaryColor] = useState("#3b36e0");
  const [isActive, setIsActive] = useState(true);

  const [uploadingLogo, setUploadingLogo] = useState(false);
  const [isSaving, startSaving] = useTransition();

  const fetchPartners = useCallback(async () => {
    setLoading(true);
    try {
      const res = await fetch("/api/admin/partners");
      const json = await res.json();
      if (json.ok) {
        setPartners(json.partners);
      }
    } catch {
      showToast("Failed to load partners", "error");
    } finally {
      setLoading(false);
    }
  }, [showToast]);

  useEffect(() => {
    fetchPartners();
  }, [fetchPartners]);

  // Live slug uniqueness check with debounce
  useEffect(() => {
    if (!drawerOpen) return;
    if (!slug) {
      setSlugStatus("idle");
      return;
    }

    if (isReservedSlug(slug)) {
      setSlugStatus("unavailable");
      setSlugReason("Slug is reserved by the system");
      return;
    }

    setSlugStatus("checking");
    const timer = setTimeout(async () => {
      try {
        const query = new URLSearchParams({ slug });
        if (editingPartner?._id) query.set("currentId", editingPartner._id);
        const res = await fetch(`/api/admin/partners/check-slug?${query.toString()}`);
        const data = await res.json();
        if (data.available) {
          setSlugStatus("available");
          setSlugReason("");
        } else {
          setSlugStatus("unavailable");
          setSlugReason(data.reason || "Slug already taken");
        }
      } catch {
        setSlugStatus("idle");
      }
    }, 350);

    return () => clearTimeout(timer);
  }, [slug, drawerOpen, editingPartner]);

  const openCreateModal = () => {
    setEditingPartner(null);
    setName("");
    setSlug("");
    setSlugStatus("idle");
    setSlugReason("");
    setTelegramLink("");
    setWhatsappLink("");
    setZoomLink("");
    setTagline("");
    setMentorBio("");
    setTestimonials([]);
    setLogoUrl("");
    setPrimaryColor("#3b36e0");
    setIsActive(true);
    setDrawerOpen(true);
  };

  const openEditModal = (partner: PartnerItem) => {
    setEditingPartner(partner);
    setName(partner.name);
    setSlug(partner.slug);
    setSlugStatus("available");
    setSlugReason("");
    setTelegramLink(partner.telegramLink);
    setWhatsappLink(partner.whatsappGroupLink || "");
    setZoomLink(partner.zoomLink || "");
    setTagline(partner.tagline || "");
    setMentorBio(partner.mentorBio || "");
    setTestimonials(partner.testimonials ?? []);
    setLogoUrl(partner.logoUrl || "");
    setPrimaryColor(partner.theme?.primaryColor || "#3b36e0");
    setIsActive(partner.isActive);
    setDrawerOpen(true);
  };

  const handleNameChange = (val: string) => {
    setName(val);
    if (!editingPartner) {
      setSlug(slugify(val));
    }
  };

  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    if (file.size > 2 * 1024 * 1024) {
      showToast("File size exceeds 2 MB limit", "error");
      return;
    }

    setUploadingLogo(true);
    try {
      const formData = new FormData();
      formData.append("file", file);

      const res = await fetch("/api/admin/upload", {
        method: "POST",
        body: formData,
      });

      const data = await res.json();
      if (res.ok && data.ok) {
        setLogoUrl(data.url);
        showToast("Logo uploaded successfully", "success");
      } else {
        showToast(data.error || "Logo upload failed", "error");
      }
    } catch {
      showToast("Network error uploading logo", "error");
    } finally {
      setUploadingLogo(false);
    }
  };

  const handleToggleActive = async (partner: PartnerItem) => {
    try {
      const res = await fetch(`/api/admin/partners/${partner._id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ isActive: !partner.isActive }),
      });
      if (res.ok) {
        setPartners((prev) =>
          prev.map((p) => (p._id === partner._id ? { ...p, isActive: !p.isActive } : p))
        );
        showToast(`Partner is now ${!partner.isActive ? "Active" : "Inactive"}`, "success");
      }
    } catch {
      showToast("Failed to toggle status", "error");
    }
  };

  const handleDelete = async () => {
    if (!deleteTarget) return;
    try {
      const res = await fetch(`/api/admin/partners/${deleteTarget._id}`, {
        method: "DELETE",
      });
      if (res.ok) {
        setPartners((prev) => prev.filter((p) => p._id !== deleteTarget._id));
        showToast("Partner deleted successfully", "success");
      } else {
        showToast("Failed to delete partner", "error");
      }
    } catch {
      showToast("Network error", "error");
    } finally {
      setDeleteTarget(null);
    }
  };

  const handleRegenerate = async () => {
    if (!regenTarget) return;
    try {
      const res = await fetch(`/api/admin/partners/${regenTarget._id}/regenerate`, { method: "POST" });
      const json = await res.json();
      if (res.ok && json.ok) {
        showToast(`New page generated for ${regenTarget.name}`, "success");
      } else {
        showToast(json.error || "Failed to regenerate page", "error");
      }
    } catch {
      showToast("Network error", "error");
    } finally {
      setRegenTarget(null);
    }
  };

  const handleSavePartner = async (e: React.FormEvent) => {
    e.preventDefault();
    if (slugStatus === "unavailable") {
      showToast(slugReason || "Please choose a valid slug", "error");
      return;
    }

    startSaving(async () => {
      try {
        const payload = {
          name: name.trim(),
          slug: slug.toLowerCase().trim(),
          telegramLink: telegramLink.trim(),
          whatsappGroupLink: whatsappLink.trim() || undefined,
          zoomLink: zoomLink.trim() || undefined,
          tagline: tagline.trim() || undefined,
          mentorBio: mentorBio.trim() || undefined,
          testimonials: testimonials
            .map((t) => ({ name: t.name.trim(), role: t.role?.trim() || undefined, quote: t.quote.trim() }))
            .filter((t) => t.name && t.quote),
          logoUrl: logoUrl.trim() || undefined,
          isActive,
          theme: { primaryColor },
        };

        const isEdit = !!editingPartner;
        const url = isEdit
          ? `/api/admin/partners/${editingPartner._id}`
          : "/api/admin/partners";
        const method = isEdit ? "PATCH" : "POST";

        const res = await fetch(url, {
          method,
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(payload),
        });

        const data = await res.json();
        if (res.ok && data.ok) {
          showToast(
            isEdit ? "Partner updated successfully!" : "Partner created successfully!",
            "success"
          );
          setDrawerOpen(false);
          await fetchPartners();

          // Set success banner showing the live URL
          const origin = window.location.origin;
          setSavedSuccessUrl(`${origin}/${data.partner.slug}`);
        } else {
          showToast(data.error || "Failed to save partner", "error");
        }
      } catch {
        showToast("Network error while saving", "error");
      }
    });
  };

  const copyToClipboard = (text: string) => {
    navigator.clipboard.writeText(text);
    showToast("URL copied to clipboard!", "success");
  };

  const getLiveUrl = (partner: PartnerItem) =>
    `${typeof window !== "undefined" ? window.location.origin : ""}/${partner.slug}`;

  /* Shared pieces used by both the mobile card list and the desktop table */
  const renderAvatar = (partner: PartnerItem): ReactNode =>
    partner.logoUrl ? (
      <img
        src={partner.logoUrl}
        alt={partner.name}
        className="size-9 shrink-0 rounded-xl border border-hairline object-contain p-0.5 bg-bg"
      />
    ) : (
      <div
        style={{ backgroundColor: partner.theme?.primaryColor || "#3b36e0" }}
        className="flex size-9 shrink-0 items-center justify-center rounded-xl text-white font-bold text-xs shadow-sm"
      >
        {partner.name.slice(0, 2).toUpperCase()}
      </div>
    );

  const renderToggle = (partner: PartnerItem): ReactNode => (
    <button
      type="button"
      role="switch"
      aria-checked={partner.isActive}
      aria-label={`${partner.name} active`}
      onClick={() => handleToggleActive(partner)}
      className={`relative inline-flex h-5 w-9 shrink-0 cursor-pointer rounded-full border-2 border-transparent transition-colors duration-200 ease-in-out focus:outline-none ${
        partner.isActive ? "bg-up" : "bg-mute/40"
      }`}
    >
      <span
        className={`pointer-events-none inline-block size-4 rounded-full bg-white shadow-md transform transition duration-200 ease-in-out ${
          partner.isActive ? "translate-x-4" : "translate-x-0"
        }`}
      />
    </button>
  );

  const renderActions = (partner: PartnerItem): ReactNode => (
    <div className="flex items-center justify-end gap-1 sm:gap-2">
      <a
        href={`/${partner.slug}`}
        target="_blank"
        rel="noopener noreferrer"
        className="rounded-lg p-2 text-mute hover:text-ink hover:bg-bg-alt"
        title="Open live partner page"
        aria-label="Open live partner page"
      >
        <svg className="size-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
          <path d="M18 13v6a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V8a2 2 0 0 1 2-2h6" />
          <polyline points="15 3 21 3 21 9" />
          <line x1="10" y1="14" x2="21" y2="3" />
        </svg>
      </a>

      <button
        onClick={() => openEditModal(partner)}
        className="rounded-lg p-2 text-mute hover:text-accent hover:bg-bg-alt"
        title="Edit partner"
        aria-label="Edit partner"
      >
        <svg className="size-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
          <path d="M17 3a2.85 2.83 0 1 1 4 4L7.5 20.5 2 22l1.5-5.5Z" />
        </svg>
      </button>

      <button
        onClick={() => setRegenTarget(partner)}
        className="rounded-lg p-2 text-mute hover:text-accent hover:bg-bg-alt"
        title="Regenerate page layout"
        aria-label={`Regenerate page for ${partner.name}`}
      >
        <Shuffle className="size-4" aria-hidden />
      </button>

      <button
        onClick={() => setDeleteTarget(partner)}
        className="rounded-lg p-2 text-mute hover:text-down hover:bg-down/10"
        title="Delete partner"
        aria-label="Delete partner"
      >
        <svg className="size-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
          <polyline points="3 6 5 6 21 6" />
          <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2" />
        </svg>
      </button>
    </div>
  );

  return (
    <div className="space-y-6">
      {/* Top Header */}
      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="text-2xl font-bold tracking-tight text-ink">Partners Management</h1>
          <p className="mt-1 text-sm text-mute">
            Create zero-redeploy partner landing pages with custom themes, links, and branding.
          </p>
        </div>

        <Button
          onClick={openCreateModal}
          className="bg-accent text-accent-ink font-semibold shadow-md shadow-accent/20"
        >
          + Add New Partner
        </Button>
      </div>

      {/* Success Live URL Banner (shown after creation/update) */}
      {savedSuccessUrl && (
        <div className="relative flex flex-col items-start justify-between gap-4 rounded-2xl border border-up/30 bg-up/10 p-4 sm:flex-row sm:items-center sm:p-5">
          <div className="flex min-w-0 items-center gap-3">
            <span className="flex size-10 shrink-0 items-center justify-center rounded-xl bg-up text-white">
              <Check className="size-5" aria-hidden />
            </span>
            <div className="min-w-0">
              <h4 className="text-sm font-bold text-ink">Partner Page is LIVE Instantly!</h4>
              <p className="mt-0.5 break-all text-xs text-mute font-mono">{savedSuccessUrl}</p>
            </div>
          </div>

          <div className="flex flex-wrap items-center gap-2">
            <button
              onClick={() => copyToClipboard(savedSuccessUrl)}
              className="inline-flex items-center gap-1.5 rounded-xl border border-hairline bg-card px-3.5 py-2 text-xs font-semibold text-ink shadow-sm transition hover:bg-bg-alt"
            >
              <Copy className="size-3.5" aria-hidden /> Copy URL
            </button>
            <a
              href={savedSuccessUrl}
              target="_blank"
              rel="noopener noreferrer"
              className="inline-flex items-center gap-1.5 rounded-xl bg-up px-3.5 py-2 text-xs font-semibold text-white shadow-sm transition hover:brightness-110"
            >
              Open Page <ExternalLink className="size-3.5" aria-hidden />
            </a>
            <button
              onClick={() => setSavedSuccessUrl(null)}
              className="p-1.5 text-mute hover:text-ink"
              aria-label="Dismiss"
            >
              <X className="size-4" aria-hidden />
            </button>
          </div>
        </div>
      )}

      {/* ───────── Mobile: card list ───────── */}
      <div className="space-y-3 md:hidden">
        {loading ? (
          <div className="rounded-2xl border border-hairline bg-card px-6 py-12 text-center text-sm text-mute">
            Loading partners...
          </div>
        ) : partners.length === 0 ? (
          <div className="rounded-2xl border border-hairline bg-card px-6 py-12 text-center text-sm text-mute">
            No partners created yet. Tap "+ Add New Partner" to launch your first partner!
          </div>
        ) : (
          partners.map((partner) => {
            const liveUrl = getLiveUrl(partner);
            return (
              <div key={partner._id} className="rounded-2xl border border-hairline bg-card p-4">
                <div className="flex items-center justify-between gap-3">
                  <div className="flex min-w-0 items-center gap-3">
                    {renderAvatar(partner)}
                    <div className="min-w-0">
                      <div className="truncate font-semibold text-ink">{partner.name}</div>
                      <div className="truncate font-mono text-xs text-mute">/{partner.slug}</div>
                    </div>
                  </div>
                  <div className="flex shrink-0 items-center gap-2">
                    <span className="text-[11px] font-semibold text-mute">
                      {partner.isActive ? "Active" : "Inactive"}
                    </span>
                    {renderToggle(partner)}
                  </div>
                </div>

                <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-2 text-xs text-mute">
                  <span className="inline-flex items-center rounded-full bg-accent/10 px-2.5 py-0.5 font-bold text-accent">
                    {partner.leadsCount.toLocaleString()} leads
                  </span>
                  <a
                    href={partner.telegramLink}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="text-accent hover:underline"
                  >
                    Telegram ↗
                  </a>
                  <span>{new Date(partner.createdAt).toLocaleDateString()}</span>
                </div>

                <div className="mt-3 flex items-center justify-between gap-2 border-t border-hairline pt-3">
                  <button
                    onClick={() => copyToClipboard(liveUrl)}
                    className="inline-flex items-center gap-1.5 rounded-lg border border-hairline px-3 py-1.5 text-xs font-semibold text-ink hover:bg-bg-alt"
                  >
                    <Copy className="size-3.5" aria-hidden /> Copy URL
                  </button>
                  {renderActions(partner)}
                </div>
              </div>
            );
          })
        )}
      </div>

      {/* ───────── Desktop: table ───────── */}
      <Card className="hidden overflow-hidden md:block">
        <div className="overflow-x-auto">
          <table className="w-full text-left text-sm">
            <thead className="border-b border-hairline bg-bg-alt/50 text-xs font-semibold uppercase tracking-wider text-mute">

            </Button>

      <ConfirmDialog
        isOpen={!!regenTarget}
        title="Regenerate Page"
        description={`Pick a brand new layout, copy, theme and section order for "${regenTarget?.name}"? The current look will be replaced.`}
        confirmText="Regenerate"
        onConfirm={handleRegenerate}
        onCancel={() => setRegenTarget(null)}
      />

      {/* Delete Confirmation Dialog */}
      <ConfirmDialog
        isOpen={!!deleteTarget}
        title="Delete Partner"
        description={`Are you sure you want to delete "${deleteTarget?.name}"? Historical leads tagged with this partner will retain their snapshot.`}
        confirmText="Delete Partner"
        danger
        onConfirm={handleDelete}
        onCancel={() => setDeleteTarget(null)}
      />
    </div>
  );
}

  const [uploadingLogo, setUploadingLogo] = useState(false);
  const [shuffleOpen, setShuffleOpen] = useState(false);

  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file || !settings) return;
    setUploadingLogo(true);
    try {
      const formData = new FormData();
      formData.append("file", file);
      const res = await fetch("/api/admin/upload", { method: "POST", body: formData });
      const data = await res.json();
      if (res.ok && data.ok) {
        setSettings((prev) => (prev ? { ...prev, logoUrl: data.url } : prev));
        showToast("Logo uploaded. Save settings to apply it.", "success");
      } else {
        showToast(data.error || "Logo upload failed", "error");
      }
    } catch {
      showToast("Network error uploading logo", "error");
    } finally {
      setUploadingLogo(false);
    }
  };

  const handleShuffle = async (original: boolean) => {
    try {
      const res = await fetch("/api/admin/settings/regenerate", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ original }),
      });
      const json = await res.json();
      if (res.ok && json.ok) {
        showToast(original ? "Root page restored to the original design" : "Root page shuffled", "success");

        showToast(json.error || "Failed to shuffle page", "error");

    fetch("/api/admin/settings")

      .then((data) => {
        if (data.ok && data.settings) {
          setSettings(data.settings);
        }
      })
      .catch(() => showToast("Failed to load settings", "error"))
      .finally(() => setLoading(false));
  }, [showToast]);

  const handleSaveSettings = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!settings) return;

    startSaving(async () => {
      try {
        const res = await fetch("/api/admin/settings", {
          method: "PUT",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(settings),
        });
        const data = await res.json();
        if (res.ok && data.ok) {

export function onSpotlightMove(e: React.PointerEvent<HTMLElement>) {
  const rect = e.currentTarget.getBoundingClientRect();
  e.currentTarget.style.setProperty("--mx", `${e.clientX - rect.left}px`);
  e.currentTarget.style.setProperty("--my", `${e.clientY - rect.top}px`);
}

/* ─────────────────────────── Button ─────────────────────────────────────── */
export type ButtonVariant = "primary" | "secondary" | "danger" | "ghost";

export interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: ButtonVariant;
  size?: "sm" | "md" | "lg";
  loading?: boolean;
  icon?: ReactNode;
}

export function Button({
  variant = "primary",
  size = "md",
  loading = false,
  icon,
  children,
  className = "",
  disabled,
  ...props
}: ButtonProps) {
  const base =
    "inline-flex items-center justify-center gap-2 whitespace-nowrap font-medium transition duration-200 active:scale-[.98] disabled:cursor-not-allowed disabled:opacity-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-accent focus-visible:ring-offset-2 focus-visible:ring-offset-card motion-reduce:transition-none";

  const sizes = {
    sm: "h-9 rounded-lg px-3.5 text-meta",
    md: "h-11 rounded-xl px-5 text-cell font-semibold",
    lg: "h-12 rounded-xl px-6 text-body font-semibold",
  };

  const variants = {
    primary:
      "bg-linear-to-br from-accent to-accent-2 text-accent-ink shadow-[0_8px_24px_-8px_color-mix(in_oklab,var(--accent)_60%,transparent)] enabled:hover:-translate-y-px enabled:hover:shadow-[0_12px_28px_-8px_color-mix(in_oklab,var(--accent)_70%,transparent)]",
    secondary:

    danger:
      "bg-down text-white shadow-[0_8px_20px_-8px_rgba(217,48,74,.5)] enabled:hover:-translate-y-px enabled:hover:bg-down/90",
    ghost:
      "text-mute hover:bg-alt hover:text-ink",
  };

  return (
    <button
      disabled={disabled || loading}
      aria-busy={loading}
      className={`${base} ${sizes[size]} ${variants[variant]} ${className}`}
      {...props}
    >
      {loading ? (

  );

            {delta.positive ? <ArrowUp className="inline size-3" aria-hidden /> : <ArrowDown className="inline size-3" aria-hidden />} {delta.value}
          </span>
        )}
      </div>
      {tooltip && (
        <p className="mt-2 text-[11px] text-faint border-t border-hairline pt-2" title={tooltip}>
          {tooltip}
        </p>
      )}
    </div>
  );
}

/* ─────────────────────────── CountUp ────────────────────────────────────── */
export function CountUp({
  value,
  duration = 1400,
  formatter,
}: {
  value: number;
  duration?: number;
  formatter?: (n: number) => string;
}) {
  const [disp, setDisp] = useState(0);
  const [started, setStarted] = useState(false);
  const ref = useRef<HTMLSpanElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el || started) return;
    const ob = new IntersectionObserver(
      ([entry]) => {
        if (entry?.isIntersecting) {
          setStarted(true);
          ob.disconnect();
        }
      },
      { threshold: 0.1 }
    );
    ob.observe(el);
    return () => ob.disconnect();
  }, [started]);

  useEffect(() => {
    if (!started) return;
    let start: number | null = null;
    const from = 0;
    const to = value;
    const step = (ts: number) => {
      if (!start) start = ts;
      const prog = Math.min((ts - start) / duration, 1);
      const ease = prog === 1 ? 1 : 1 - Math.pow(2, -10 * prog);
      setDisp(Math.round(from + (to - from) * ease));
      if (prog < 1) requestAnimationFrame(step);
    };
    const req = requestAnimationFrame(step);
    return () => cancelAnimationFrame(req);
  }, [started, value, duration]);

  const fmt = formatter ?? ((n: number) => n.toLocaleString());
  return (
    <span ref={ref} className="tabular-nums">
      {fmt(started ? disp : 0)}
    </span>
  );
}

/* ─────────────────────────── Custom Select ──────────────────────────────── */
export interface SelectOption {
  value: string;
  label: string;
  icon?: ReactNode;
}

export function Select({
  value,
  options,
  onChange,
  placeholder = "Select...",
  disabled = false,
  className = "",
}: {
  value: string;
  options: SelectOption[];
  onChange: (val: string) => void;
  placeholder?: string;
  disabled?: boolean;
  className?: string;
}) {
  const [open, setOpen] = useState(false);
  const [focusIndex, setFocusIndex] = useState(-1);
  const containerRef = useRef<HTMLDivElement>(null);
  const listboxId = useId();

  const selected = options.find((o) => o.value === value);

  useEffect(() => {
    const onOutside = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setOpen(false);
      }
    };
    document.addEventListener("mousedown", onOutside);
    return () => document.removeEventListener("mousedown", onOutside);
  }, []);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (disabled) return;
    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      if (open && focusIndex >= 0 && options[focusIndex]) {
        onChange(options[focusIndex].value);
        setOpen(false);
      } else {
        setOpen(!open);
      }
    } else if (e.key === "Escape") {
      setOpen(false);
    } else if (e.key === "ArrowDown") {
      e.preventDefault();
      if (!open) setOpen(true);
      else setFocusIndex((prev) => (prev < options.length - 1 ? prev + 1 : 0));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      if (!open) setOpen(true);
      else setFocusIndex((prev) => (prev > 0 ? prev - 1 : options.length - 1));
    }
  };

  return (
    <div ref={containerRef} className={`relative ${className}`}>
      <button
        type="button"
        role="combobox"
        aria-expanded={open}
        aria-controls={open ? listboxId : undefined}
        aria-haspopup="listbox"
        disabled={disabled}
        onClick={() => setOpen(!open)}
        onKeyDown={handleKeyDown}
        className="flex h-11 w-full items-center justify-between rounded-xl border border-hairline-strong bg-bg px-3.5 text-cell text-ink transition hover:border-accent/60 focus-visible:border-accent focus-visible:ring-2 focus-visible:ring-accent/30 disabled:opacity-50"
      >
        <span className="flex items-center gap-2 truncate">
          {selected?.icon}
          <span className={selected ? "text-ink" : "text-faint"}>{selected ? selected.label : placeholder}</span>
        </span>
        <svg
          className={`size-4 text-mute transition-transform duration-200 ${open ? "rotate-180" : ""}`}
          viewBox="0 0 24 24"
          fill="none"
          stroke="currentColor"
          strokeWidth="2"
        >
          <path d="m6 9 6 6 6-6" />
        </svg>
      </button>

      {open && (
        <div
          id={listboxId}
          role="listbox"
          className="absolute z-50 mt-1 max-h-60 w-full overflow-auto rounded-xl border border-hairline bg-card p-1 shadow-xl animate-[f-fadein_.15s_ease-out]"
        >
          {options.map((opt, i) => {
            const isSel = opt.value === value;
            const isFocused = i === focusIndex;
            return (
              <div
                key={opt.value}
                role="option"
                aria-selected={isSel}
                onClick={() => {
                  onChange(opt.value);
                  setOpen(false);
                }}
                className={`flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-cell transition ${
                  isSel
                    ? "bg-accent/10 font-semibold text-accent"
                    : isFocused
                    ? "bg-alt text-ink"
                    : "text-ink hover:bg-alt"
                }`}
              >
                <div className="flex items-center gap-2">
                  {opt.icon}
                  <span>{opt.label}</span>
                </div>
                {isSel && (
                  <svg className="size-4 text-accent" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
                    <path d="M20 6 9 17l-5-5" />
                  </svg>
                )}
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}

/* ─────────────────────────── DateTimeField ──────────────────────────────── */
// Timezone safe helpers:
export function toLocalInput(iso?: string | null): string {
  if (!iso) return "";
  const d = new Date(iso);
  if (isNaN(d.getTime())) return "";
  const pad = (n: number) => String(n).padStart(2, "0");
  const year = d.getFullYear();
  const month = pad(d.getMonth() + 1);
  const day = pad(d.getDate());
  const hours = pad(d.getHours());
  const minutes = pad(d.getMinutes());
  return `${year}-${month}-${day}T${hours}:${minutes}`;
}

export function fromLocalInput(localVal: string): string | null {
  if (!localVal) return null;
  const d = new Date(localVal);
  return isNaN(d.getTime()) ? null : d.toISOString();
}

export function DateTimeField({
  value,
  onChange,
  disabled = false,
}: {
  value: string;
  onChange: (iso: string) => void;
  disabled?: boolean;
}) {
  const localVal = toLocalInput(value);
  const tzName = Intl.DateTimeFormat().resolvedOptions().timeZone;

  return (
    <div className="space-y-1.5">
      <div className="relative">
        <input
          type="datetime-local"
          value={localVal}
          disabled={disabled}
          onChange={(e) => {
            const iso = fromLocalInput(e.target.value);
            if (iso) onChange(iso);
          }}
          className="h-11 w-full rounded-xl border border-hairline-strong bg-bg px-3.5 text-cell text-ink outline-none transition focus-visible:border-accent focus-visible:ring-2 focus-visible:ring-accent/30 disabled:opacity-50"
        />
      </div>
      <div className="flex items-center justify-between text-[11px] text-faint">
        <span>Browser Zone: {tzName}</span>
        <span>Stored UTC: {value ? new Date(value).toISOString() : "Not set"}</span>
      </div>
    </div>
  );
}

  description,

  cancelLabel = "Cancel",

  confirmPhrase,
  variant = "danger",
  danger,
  loading = false,
  onConfirm,

  onClose,

"use client";

import { useEffect, useRef } from "react";

export function Confetti() {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    let animId: number;
    const width = (canvas.width = canvas.parentElement?.clientWidth || window.innerWidth);
    const height = (canvas.height = canvas.parentElement?.clientHeight || 400);

    const colors = ["#3b36e0", "#7a5cff", "#0f8f4f", "#f2b33d", "#ec4899", "#06b6d4"];
    const pieces: {
      x: number;
      y: number;
      size: number;
      color: string;
      speedX: number;
      speedY: number;
      rotation: number;
      rotationSpeed: number;
      opacity: number;
    }[] = [];

    for (let i = 0; i < 70; i++) {
      pieces.push({
        x: width * 0.5 + (Math.random() - 0.5) * 80,
        y: height * 0.4 + (Math.random() - 0.5) * 50,
        size: Math.random() * 8 + 4,
        color: colors[Math.floor(Math.random() * colors.length)],
        speedX: (Math.random() - 0.5) * 12,
        speedY: (Math.random() - 1.2) * 10 - 2,
        rotation: Math.random() * 360,
        rotationSpeed: (Math.random() - 0.5) * 10,
        opacity: 1,
      });
    }

    let frame = 0;
    const render = () => {
      frame++;
      ctx.clearRect(0, 0, width, height);

      let activePieces = 0;
      for (const p of pieces) {
        p.x += p.speedX;
        p.y += p.speedY;
        p.speedY += 0.28; // gravity
        p.speedX *= 0.98; // air drag
        p.rotation += p.rotationSpeed;
        if (frame > 45) {
          p.opacity -= 0.015;
        }

        if (p.opacity > 0) {
          activePieces++;
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate((p.rotation * Math.PI) / 180);
          ctx.fillStyle = p.color;
          ctx.globalAlpha = Math.max(0, p.opacity);
          ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size * 0.6);
          ctx.restore();
        }
      }

      if (activePieces > 0) {
        animId = requestAnimationFrame(render);
      }
    };

    animId = requestAnimationFrame(render);

    return () => {
      cancelAnimationFrame(animId);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="pointer-events-none absolute inset-0 z-20 h-full w-full"
    />
  );
}

function Select<T extends string>({ label, value, options, onChange }: { label: string; value: T; options: readonly T[]; onChange: (v: T) => void }) {

    <label className="flex flex-col gap-1 text-[11px] font-semibold text-mute">

        value={value}
        onChange={(e) => onChange(e.target.value as T)}

"use client";

import type { CSSProperties } from "react";
import type { SiteSettingsData } from "@/lib/defaults";
import { generatePartnerPage } from "@/lib/page-builder/generate";
import type { SectionKey } from "@/lib/page-builder/types";
import { AnimationProvider, useAnimationPreset } from "./motion";
import { resolvePage, type PagePlan, type PartnerView } from "./page-context";
import { RenderSection } from "./registry";

interface LandingPageProps {
  partner?: PartnerView | null;
  settings: SiteSettingsData;
  isCustomPartner?: boolean;
  /** The partner's saved page. Omitted for the root page, which renders the original design. */
  page?: PagePlan | null;
  /** Dev preview: renders the lead form as an inert placeholder. */
  preview?: boolean;
}

/** The original design: variant 1 and the original copy for every section. */
const ORIGINAL_PAGE: PagePlan = generatePartnerPage({}, { legacy: true, seed: 0 });

function PageBody({ page, plan }: { page: ReturnType<typeof resolvePage>; plan: PagePlan }) {
  const preset = useAnimationPreset();
  const style = page.themeVars as CSSProperties;
  const { order, content, ctx } = page;
  const render = (k: SectionKey) => (
    <RenderSection key={k} name={k} variantId={plan.layoutConfig[k]} content={content[k]} ctx={ctx} />
  );

  return (
    <div
      className="min-h-screen bg-bg text-ink transition-colors duration-300"
      style={style}
      data-btn={plan.themeConfig.buttonStyle}
      data-type={plan.themeConfig.typeStyle}
      data-anim={preset.style}
      data-lift={String(preset.hoverLift)}
    >
      {/* Background glow blobs */}
      <div className="pointer-events-none fixed inset-0 overflow-hidden">
        <div className="f-blob a" />
        <div className="f-blob b" />
        <div className="f-blob c" />
      </div>

      {order.filter((k) => k === "navbar").map(render)}
      <main className="relative z-10">{order.filter((k) => k !== "navbar" && k !== "footer").map(render)}</main>
      {order.filter((k) => k === "footer").map(render)}
    </div>
  );
}

/** Renders a partner's saved page (or the original design) through the section registry. */
export function LandingPage({ partner, settings, isCustomPartner = false, page, preview = false }: LandingPageProps) {
  const plan = page ?? ORIGINAL_PAGE;
  const resolved = resolvePage({ partner, settings, isCustomPartner, plan, preview });
  return (
    <AnimationProvider style={plan.themeConfig.animationStyle}>
      <PageBody page={resolved} plan={plan} />
    </AnimationProvider>
  );
}

"use client";

import { Fragment, type ElementType } from "react";
import { useAnimationPreset } from "./context";
import { useRevealState, usePrefersReducedMotion } from "./hooks";
import type { CssVars } from "./Reveal";
import type { TextKind } from "./presets";

interface AnimatedTextProps {
  text: string;
  as?: ElementType;
  className?: string;
  /** Overrides the preset's text effect. */
  mode?: TextKind;
  /** Above-the-fold: play on first paint. */
  eager?: boolean;
  delay?: number;
  id?: string;
}

const LETTER_MODES: TextKind[] = ["typewriter", "blur-letters"];
const MAX_TOTAL_DELAY = 700;

/**
 * Word-by-word reveal, mask slide-up, typewriter or blur-in letter stagger.
 * The full text is always present for screen readers and before hydration.
 */
export function AnimatedText({ text, as, className = "", mode, eager = false, delay = 0, id }: AnimatedTextProps) {
  const Tag: ElementType = as ?? "span";
  const preset = useAnimationPreset();
  const reduced = usePrefersReducedMotion();
  const kind = mode ?? preset.text;
  const enabled = kind !== "none" && !reduced;
  const [ref, state] = useRevealState<HTMLElement>({ enabled, eager });

  if (!enabled) {
    return (
      <Tag id={id} className={className}>
        {text}
      </Tag>
    );
  }

  const byLetter = LETTER_MODES.includes(kind);
  const words = text.split(/\s+/).filter(Boolean);
  // Index of each word's first animated unit (letters also count the space after a word).
  const starts = words.reduce<number[]>((acc, _w, i) => {
    acc.push(i === 0 ? 0 : acc[i - 1] + (byLetter ? Array.from(words[i - 1]).length + 1 : 1));
    return acc;
  }, []);
  const total = words.length === 0 ? 0 : starts[words.length - 1] + (byLetter ? Array.from(words[words.length - 1]).length : 1);
  const step = Math.min(preset.textStep, MAX_TOTAL_DELAY / Math.max(total, 1));
  const delayOf = (unit: number) => ({ "--d": `${delay + unit * step}ms` }) as CssVars;

  return (
    <Tag ref={ref} id={id} className={className}>
      <span className="sr-only">{text}</span>
      <span
        aria-hidden="true"
        className="rt-text"
        data-mode={kind}
        data-state={state}
        style={{ "--rt-dur": `${preset.duration}ms` } as CssVars}
      >
        {words.map((word, wi) => (
          <Fragment key={wi}>
            <span className="rt-wrap">
              {byLetter ? (
                Array.from(word).map((ch, ci) => (
                  <span key={ci} className="rt-u" style={delayOf(starts[wi] + ci)}>
                    {ch}
                  </span>
                ))
              ) : (
                <span className="rt-u" style={delayOf(starts[wi])}>
                  {word}
                </span>
              )}
            </span>
            {wi < words.length - 1 ? " " : null}
          </Fragment>
        ))}
      </span>
    </Tag>
  );
}

/** Moves its child slightly slower than the scroll. No-op unless the preset enables parallax. */
export function Parallax({ children, speed = 0.12, className = "" }: { children: ReactNode; speed?: number; className?: string }) {
  const preset = useAnimationPreset();
  const reduced = usePrefersReducedMotion();
  const ref = useRef<HTMLDivElement>(null);
  const enabled = preset.parallax && !reduced;

  useEffect(() => {
    const el = ref.current;
    if (!enabled || !el) return;
    let frame = 0;
    const update = () => {
      frame = 0;
      const rect = el.getBoundingClientRect();
      const offset = (rect.top + rect.height / 2 - window.innerHeight / 2) * speed;
      el.style.transform = `translate3d(0, ${(-offset).toFixed(1)}px, 0)`;
    };
    const onScroll = () => {
      if (!frame) frame = requestAnimationFrame(update);
    };
    update();
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", onScroll);
    return () => {
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", onScroll);
      if (frame) cancelAnimationFrame(frame);
    };
  }, [enabled, speed]);

  return (
    <div ref={ref} className={className}>
      {children}
    </div>
  );
}

export type CssVars = CSSProperties & Record<`--${string}`, string | number | undefined>;

interface RevealProps {
  children: ReactNode;
  as?: ElementType;
  className?: string;
  /** Sibling position: delays the reveal by `index * preset.stagger`. */
  index?: number;
  /** Extra delay in ms. */
  delay?: number;
  /** Above-the-fold content: animate on first paint instead of on scroll. */
  eager?: boolean;
  id?: string;
}

/** Scroll-triggered fade / slide / blur reveal chosen by the page's animation preset. */
export function Reveal({ children, as, className = "", index = 0, delay = 0, eager = false, id }: RevealProps) {
  const Tag: ElementType = as ?? "div";
  const preset = useAnimationPreset();
  const reduced = usePrefersReducedMotion();
  const enabled = preset.reveal !== "none" && !reduced;
  const [ref, state] = useRevealState<HTMLElement>({ enabled, eager });

  const cls =
    state === "idle" ? "" : eager ? "rt-reveal-eager" : state === "shown" ? "rt-reveal is-in" : "rt-reveal";
  const style: CssVars = {
    "--rt-dist": `${preset.distance}px`,
    "--rt-dur": `${preset.duration}ms`,
    "--rt-blur": preset.reveal === "blur" ? "10px" : "0px",
    "--rt-delay": `${delay + index * preset.stagger}ms`,
  };

  return (
    <Tag ref={ref} id={id} className={`${cls} ${className}`.trim()} style={enabled ? style : undefined}>

function StatValue({ item }: { item: StatItem }) {
  if (item.end === undefined) return <>{item.text ?? ""}</>;
  return <CountUp end={item.end} decimals={item.decimals} suffix={item.suffix} />;
}

function staticValue(item: StatItem): string {
  if (item.end === undefined) return item.text ?? "";
  const n = item.end.toLocaleString("en-US", { minimumFractionDigits: item.decimals ?? 0, maximumFractionDigits: item.decimals ?? 0 });
  return `${n}${item.suffix ?? ""}`;
}

/** Themed primary button (look is driven by theme_config.buttonStyle through data-btn on the page root). */
export function CtaButton({
  children,
  className = "",
  onClick,
  href,
  external = false,
}: {
  children: ReactNode;
  className?: string;
  onClick?: () => void;
  href?: string;
  external?: boolean;
}) {
  const cls = `rt-btn ${className}`.trim();
  if (href) {
    return (
      <a href={href} className={cls} {...(external ? { target: "_blank", rel: "noopener noreferrer" } : {})}>
        {children}
      </a>
    );
  }
  return (
    <button type="button" onClick={onClick ?? scrollToForm} className={cls}>
      {children}
    </button>
  );
}

  );
}

export function initials(name: string): string {
  return name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((p) => p[0]?.toUpperCase() ?? "")
    .join("");
}

"use client";

import { useEffect, useState, useRef } from "react";
import { usePrefersReducedMotion } from "./motion/hooks";

export function CountUp({
  end,
  duration = 2000,
  suffix = "",
  decimals = 0,
}: {
  end: number;
  duration?: number;
  suffix?: string;
  decimals?: number;
}) {
  const [val, setVal] = useState(0);
  const ref = useRef<HTMLSpanElement>(null);
  const [hasStarted, setHasStarted] = useState(false);
  const reducedMotion = usePrefersReducedMotion();

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setHasStarted(true);
          observer.disconnect();
        }
      },
      { threshold: 0.1 }
    );

    if (ref.current) observer.observe(ref.current);
    return () => observer.disconnect();
  }, []);

  useEffect(() => {
    if (!hasStarted) return;
    if (reducedMotion) {
      setVal(end);
      return;
    }
    let startTimestamp: number | null = null;
    let frameId: number;

    const step = (timestamp: number) => {
      if (!startTimestamp) startTimestamp = timestamp;
      const progress = Math.min((timestamp - startTimestamp) / duration, 1);
      // easeOutExpo
      const ease = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
      setVal(ease * end);

      if (progress < 1) {
        frameId = requestAnimationFrame(step);
      }
    };

    frameId = requestAnimationFrame(step);
    return () => cancelAnimationFrame(frameId);
  }, [hasStarted, end, duration, reducedMotion]);

  return (
    <span ref={ref}>
      {val.toLocaleString("en-US", {
        minimumFractionDigits: decimals,
        maximumFractionDigits: decimals,
      })}
      {suffix}
    </span>
  );
}

export function Sparkline({ values }: { values: number[] }) {
  if (values.length < 2) return null;
  const min = Math.min(...values);
  const max = Math.max(...values);
  const range = max - min || 1;
  const w = 100;
  const h = 28;

  const points = values
    .map((v, i) => {
      const x = (i / (values.length - 1)) * w;
      const y = h - ((v - min) / range) * h;
      return `${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(" ");

  return (
    <svg viewBox={`0 0 ${w} ${h}`} className="h-7 w-24 shrink-0 overflow-visible">
      <polyline
        fill="none"
        stroke="var(--up)"
        strokeWidth="2"
        strokeLinecap="round"
        strokeLinejoin="round"
        points={points}
      />
    </svg>
  );
}
